import React from "react";
import { motion } from "framer-motion";

// Skill Card Component
const SkillCard = ({ category, tools }) => {
    return (
        <motion.div
            className="w-56 h-auto p-4 space-y-2 bg-light dark:bg-dark rounded-lg shadow-md hover:shadow-lg transition duration-300 group mx-4 border-2 border-black dark:border-gray-700"
            whileHover={{ scale: 1.05 }}
        >
            {/* Category Title */}
            <h3 className="text-lg font-semibold text-dark dark:text-light group-hover:text-dark/80 group-hover:dark:text-light/80 break-words">
                {category}
            </h3>

            {/* Tool List (Initially hidden, shown on hover) */}
            <ul className="space-y-2 text-dark/75 dark:text-light/75 opacity-0 group-hover:opacity-100 max-h-0 group-hover:max-h-60 overflow-hidden transition-all duration-300 ease-in-out">
                {tools.map((tool, index) => (
                    <li key={index} className="text-sm font-medium break-words">
                        {tool}
                    </li>
                ))}
            </ul>
        </motion.div>
    );
};

const Skills = () => {
    const skillData = [
        { category: "AI", tools: ["Machine Learning", "Deep Learning", "Natural Language Processing"] },
        { category: "Web Development", tools: ["HTML", "CSS", "JavaScript", "ReactJS", "NextJS", "Tailwind CSS"] },
        { category: "Web Design", tools: ["Figma", "Adobe XD", "Wireframing", "Prototyping"] },
        { category: "Cloud & Hosting", tools: ["Firebase", "Vercel", "Netlify"] },
        { category: "Databases", tools: ["MongoDB", "MySQL", "PostgreSQL"] },
    ];

    return (
        <>
            <div className="w-full flex flex-wrap justify-start gap-6">
                {/* Side-by-Side Layout for Skill Categories */}
                {skillData.map((skill, index) => (
                    <SkillCard key={index} category={skill.category} tools={skill.tools} />
                ))}
            </div>
        </>
    );
};

export default Skills;
